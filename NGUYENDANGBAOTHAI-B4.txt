using System;

Console.OutputEncoding = System.Text.Encoding.UTF8;
int choice;

do
{
    Console.Clear();
    Console.WriteLine("=== MENU CHƯƠNG TRÌNH ===");
    Console.WriteLine("1. Bài 1: Máy tính bỏ túi");
    Console.WriteLine("2. Bài 2: Giải phương trình bậc 2");
    Console.WriteLine("3. Bài 3: Số nguyên tố & Fibonacci");
    Console.WriteLine("0. Thoát");
    Console.Write("Chọn (0-3): ");
    
    int.TryParse(Console.ReadLine(), out choice);
    Console.Clear();

    switch (choice)
    {
        case 1:
            Console.Write("Nhập a, b: ");
            double a1 = double.Parse(Console.ReadLine()!), b1 = double.Parse(Console.ReadLine()!);
            Console.Write("Phép tính (+,-,*,/,%): ");
            char op = char.Parse(Console.ReadLine()!);
            string res = (op, b1) switch {
                ('+', _) => $"{a1 + b1:F2}", ('-', _) => $"{a1 - b1:F2}",
                ('*', _) => $"{a1 * b1:F2}", ('/', 0) or ('%', 0) => "Lỗi chia cho 0!",
                ('/', _) => $"{a1 / b1:F2}", ('%', _) => $"{a1 % b1:F2}", _ => "Lỗi!"
            };
            Console.WriteLine($"Kết quả: {res}");
            break;

        case 2:
            Console.Write("Nhập a, b, c: ");
            double a2 = double.Parse(Console.ReadLine()!), b2 = double.Parse(Console.ReadLine()!), c2 = double.Parse(Console.ReadLine()!);
            if (a2 == 0) Console.WriteLine(b2 == 0 ? (c2 == 0 ? "Vô số nghiệm" : "Vô nghiệm") : $"x = {-c2/b2:F2}");
            else {
                double d = b2 * b2 - 4 * a2 * c2;
                if (d < 0) Console.WriteLine("Vô nghiệm");
                else if (d == 0) Console.WriteLine($"Nghiệm kép x = {-b2/(2*a2):F2}");
                else Console.WriteLine($"x1 = {(-b2 + Math.Sqrt(d))/(2*a2):F2}, x2 = {(-b2 - Math.Sqrt(d))/(2*a2):F2}");
            }
            break;

        case 3:
            Console.Write("Nhập N: ");
            int n = int.Parse(Console.ReadLine()!);
            bool p = n >= 2; for (int i = 2; i <= Math.Sqrt(n); i++) if (n % i == 0) p = false;
            int sum = 0; for (int i = 1; i <= n / 2; i++) if (n % i == 0) sum += i;
            Console.WriteLine($"{n} {(sum == n && n > 1 ? "là" : "KHÔNG")} số hoàn hảo, {(p ? "là" : "KHÔNG")} số nguyên tố.");
            Console.Write($"Fibonacci {n} số: ");
            long f1 = 0, f2 = 1;
            for (int i = 0; i < n; i++) { Console.Write(f1 + " "); long nx = f1 + f2; f1 = f2; f2 = nx; }
            Console.WriteLine();
            break;
    }

    if (choice != 0) { Console.WriteLine("\nBấm phím bất kỳ để về Menu..."); Console.ReadKey(); }
} while (choice != 0);
